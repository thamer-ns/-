    require(source, "float horizonFactor = chartSeconds <= 60 ? 1.00", "granular target horizon")
    require(source, "f_stop_atr_limits(chartSeconds)", "granular stop horizon")
    require(source, "array.from(fibObjective1, retracementTargetLevel)", "retracement plus extension")
    require(source, "array.from(historicalPivotObstacle, horizontalObstacleLevel, opposingZoneBoundary, targetWyckoffLevel, minorChannelBoundaryTarget, majorChannelBoundaryTarget)", "structural hard obstacles")

        60: (1.00, (0.60, 2.50)),

def test_monotonic_target_geometry() -> None:
    # The fallback geometry used when no ranked T2/T3 exists must keep direction.
    for is_long in (True, False):
        entry = 100.0
        risk = 2.0
        horizon = 1.75
        rr1, rr2, rr3 = 1.0, 2.0, 3.0
        d1 = risk * max(rr1 * horizon, 1.0)
        d2 = risk * max(rr2 * horizon, max(rr1 * horizon, 1.0) + 0.75)
        d3 = risk * max(rr3 * horizon, max(rr2 * horizon, max(rr1 * horizon, 1.0) + 0.75) + 1.0)
        targets = [entry + d1, entry + d2, entry + d3] if is_long else [entry - d1, entry - d2, entry - d3]
        if is_long:
            assert entry < targets[0] < targets[1] < targets[2]
        else:
            assert entry > targets[0] > targets[1] > targets[2]
